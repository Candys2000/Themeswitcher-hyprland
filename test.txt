hl.bind(mainMod .. " + b", function ()
  -- check if waybar is currently running
  local handle = io.popen("pgrep -x waybar")
  local result = handle:read("*a")
  handle:close()

  if result ~= "" then
    -- if running close it
    hl.dsp.exec_cmd("pkill -SIGUSR1 waybar && waybar -c ~/.config/waybar/config.json -s ~/.config/waybar/style.css)
  else
    -- if not running launch it in the background or smth
    hl.dsp.exec_cmd("waybar &&")
  end
end)
